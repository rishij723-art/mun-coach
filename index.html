    // --- Storage helpers

    const save = (k, v) => localStorage.setItem(k, v);

    const load = (k, f="") => localStorage.getItem(k) ?? f;


    // --- System prompt

    const DEFAULT_SYSTEM_PROMPT = \`You are "MUN Coach", an AI prep & research assistant for Model UN.

- Be neutral and fact-focused; separate facts from inferences.

- For country positions: prefer recent, verifiable sources; flag uncertainty.

- Output format: (1) TL;DR bullets (3–6) (2) Main response (structured) (3) Fact Check & Sources (queries/treaties/keywords).

- Provide bloc maps and SMART clauses when asked.

- Avoid legal advice and hallucinations; if unsure, say so and suggest how to verify.\`;


    // --- Presets

    const PRESETS = [

      { key:"country-brief", title:"Country Brief", hint:"1-page stance + bloc map",

        prompt:({country="India", topic="AI governance in armed conflict"}) =>

          \`Create a concise 1-page country brief on \${country} for: "\${topic}".

Include: core interests; likely stance; historical votes & treaties; key ministries; red lines; 3–5 talking points; bloc map (allies/swing/opponents) w/ 1-line rationale each.\` },

      { key:"topic-primer", title:"Topic Primer", hint:"Key defs, actors, levers",

        prompt:({topic="Water security and transboundary rivers"}) =>

          \`Make a topic primer for "\${topic}". Define terms, list major UN bodies/NGOs, outline 4–6 policy levers, and summarize 3 recent developments (with dates). End with 6 search queries to verify.\` },

      { key:"speech-coach", title:"Opening Speech (60–90s)", hint:"Hook + stance + asks",

        prompt:({country="Kenya", topic="Cybersecurity norms"}) =>

          \`Write a 60–90 second opening speech for \${country} on "\${topic}". Structure: Hook · Problem · 2–3 concrete policies · Asks · Memorable close. Add 3 POIs and strong answers.\` },

      { key:"resolution-builder", title:"Resolution Builder", hint:"SMART clauses + ops",

        prompt:({committee="UNGA", topic="Sustainable urban housing"}) =>

          \`Draft 6–10 operative clauses for a \${committee} resolution on "\${topic}". Each must be SMART and include funding source, responsible body, timeline, and monitoring.\` },

      { key:"bloc-strategy", title:"Bloc Strategy", hint:"Allies, trades, timeline",

        prompt:({country="Brazil", topic="Deforestation & supply chains"}) =>

          \`Design a bloc strategy for \${country} on "\${topic}": target allies + trades; a 4-phase lobbying timeline; 3 red lines; whip list.\` },

      { key:"practice-round", title:"Practice Round", hint:"Chair + POIs + feedback",

        prompt:({country="France", topic="Space debris mitigation"}) =>

          \`Run a 5-minute practice round for \${country} on "\${topic}": 60s mini-speech, 4 POIs to ask me, then delivery/strategy feedback.\` },

    ];


    // --- DOM refs

    const baseUrl = document.getElementById('baseUrl');

    const apiKey = document.getElementById('apiKey');

    const model = document.getElementById('model');

    const offline = document.getElementById('offline');

    const systemPrompt = document.getElementById('systemPrompt');

    const resetPrompt = document.getElementById('resetPrompt');


    const presetButtons = document.getElementById('presetButtons');

    const input = document.getElementById('input');

    const send = document.getElementById('send');

    const clearBtn = document.getElementById('clear');

    const chat = document.getElementById('chat');

    const modeBadge = document.getElementById('modeBadge');

    const hint = document.getElementById('hint');


    // --- init

    baseUrl.value = load('mun_base_url','https://api.openai.com');

    apiKey.value = load('mun_api_key','');

    model.value = load('mun_model','gpt-4o-mini');

    systemPrompt.value = load('mun_system', DEFAULT_SYSTEM_PROMPT);

    offline.checked = !apiKey.value;


    baseUrl.oninput = () => save('mun_base_url', baseUrl.value);

    apiKey.oninput = () => { save('mun_api_key', apiKey.value); offline.checked = !apiKey.value; };

    model.onchange = () => save('mun_model', model.value);

    systemPrompt.oninput = () => save('mun_system', systemPrompt.value);

    offline.onchange = () => {};


    resetPrompt.onclick = () => { systemPrompt.value = DEFAULT_SYSTEM_PROMPT; save('mun_system', systemPrompt.value); };


    // Preset buttons

    PRESETS.forEach(p => {

      const b = document.createElement('button');

      b.className = "border bg-slate-50 hover:bg-slate-100 rounded-xl px-3 py-2 text-left text-sm";

      b.textContent = p.title;

      b.onclick = () => {

        input.value = p.prompt({});

        modeBadge.textContent = p.key;

        hint.textContent = p.hint;

      };

      presetButtons.appendChild(b);

    });


    // Chat state

    let messages = [];


    function render() {

      chat.innerHTML = '';

      if (messages.length === 0) {

        const empty = document.createElement('div');

        empty.className = "h-[20vh] flex items-center justify-center text-slate-500 text-sm";

        empty.textContent = "Start the conversation above — or pick a preset on the left.";

        chat.appendChild(empty);

        return;

      }

      messages.forEach(m => {

        const row = document.createElement('div');

        row.className = "flex " + (m.role === 'user' ? 'justify-end' : 'justify-start');

        const bubble = document.createElement('div');

        bubble.className = "bubble " + (m.role === 'user' ? 'bg-slate-900 text-white' : 'bg-slate-100 text-slate-900');

        bubble.style.maxWidth = '85%';

        bubble.innerText = m.content;

        row.appendChild(bubble);

        chat.appendChild(row);

      });

      chat.scrollTop = chat.scrollHeight;

    }


    function offlineReply(user, mode) {

      const lower = user.toLowerCase();

      if (mode === "country-brief") {

        return `TL;DR

- Core interests: security, growth, alliances

- Likely stance: pragmatic cooperation

- Bloc: allies/swing/opponents


Main

• Interests: energy security; trade access; regional stability.

• Stance: balanced framework; cautious enforcement.

• History/Treaties: check UN votes 2015–2024; ratifications.

• Red Lines: sovereignty, budget impact.

• Talking Points: 1) capacity-building 2) data-sharing hub 3) targeted funding.


Fact Check & Sources

- site:un.org <topic> <country> voting record

- UNTC; latest GA/SC documents.`;

      }

      if (mode === "speech-coach") {

        return `TL;DR

- Hook → stance → 2–3 policies → asks → close


Main

Honourable Chair, distinguished delegates—our delegation proposes:

1) Coordinated standards; 2) Capacity-building fund; 3) Independent monitoring.


POIs

1) How will you finance the fund?

2) How do you balance sovereignty and oversight?

3) What metrics define success?


Fact Check & Sources

- UN ODS; recent GA resolutions; SDG indicators.`;

      }

      if (lower.includes("resolution") || mode === "resolution-builder") {

        return `TL;DR

- 6–10 SMART clauses with funding & oversight


Main (sample)

1) Requests UNDP to establish a pilot fund (USD 25M) within 12 months; report biannually to ECOSOC.

2) Urges Member States to file national roadmaps within 9 months; Secretariat dashboard.


Fact Check & Sources

- Past resolutions; agency mandates; budget comps.`;

      }

      if (lower.includes("bloc") || mode === "bloc-strategy") {

        return `TL;DR

- Targets, trades, timeline, whip list


Main

• Allies: A, B (trade wording + funding)

• Swing: C, D (pilot seats)

• Opponents: E (privacy), F (budget)

• Timeline: caucus → drafting → whip → amendments.


Fact Check & Sources

- Past votes; mission statements; regional briefs.`;

      }

      return `TL;DR

- Structured answer with steps and verification queries.


Main

• Define terms, list actors, propose levers tailored to the committee.

• Provide 3–6 bullets, then the full text.


Fact Check & Sources

- Search: site:un.org, ODS, country mission pages, World Bank data.`;

    }


    async function chatComplete({ messages, temperature=0.4 }) {

      const url = (baseUrl.value || '').replace(/\/$/, '') + '/v1/chat/completions';

      const res = await fetch(url, {

        method: 'POST',

        headers: {

          'Content-Type':'application/json',

          'Authorization': 'Bearer ' + apiKey.value,

        },

        body: JSON.stringify({

          model: model.value,

          messages: messages.map(({role, content}) => ({ role, content })),

          temperature,

        }),

      });

      if (!res.ok) throw new Error('API error ' + res.status + ': ' + (await res.text()));

      const data = await res.json();

      return data?.choices?.[0]?.message?.content || '';

    }


    send.onclick = async () => {

      const text = input.value.trim();

      if (!text) return;

      const mode = modeBadge.textContent === 'Free mode' ? 'free' : modeBadge.textContent;

      messages.push({ role: 'user', content: text, mode });

      input.value = '';

      render();


      try {

        let reply = '';

        if (!apiKey.value || offline.checked) {

          reply = offlineReply(text, mode);

        } else {

          const sys = { role: 'system', content: systemPrompt.value };

          reply = await chatComplete({ messages: [sys, ...messages] });

        }

        messages.push({ role:'assistant', content: reply, mode });

        render();

      } catch (err) {

        messages.push({ role:'assistant', content: 'Error calling API: ' + (err?.message || err) });

        render();

      }

    };


    clearBtn.onclick = () => { messages = []; render(); };


    // initial render

    render();

  </script>
